using System.Collections; using System.Collections.Generic; using UnityEngine; using UnityEngine.UI; namespace JoyBitStudios.Core { // ========================================== // 1. SISTEMA BASE DE INTERACCIÓN Y ENTORNO // ========================================== public interface IInteractable { void Interact(PlayerController player); string GetInteractionText(); } public class PlayerController : MonoBehaviour { public float moveSpeed = 5f; public int gold = 0; public int magicWood = 0; public bool hasAxe = false; public List inventoryKeys = new List(); public Text uiTextDisplay; private Rigidbody2D rb; private Vector2 moveInput; void Start() { rb = GetComponent(); if (uiTextDisplay != null) uiTextDisplay.text = ""; } void Update() { moveInput.x = Input.GetAxisRaw("Horizontal"); moveInput.y = Input.GetAxisRaw("Vertical"); // Interacción manual con la tecla E if (Input.GetKeyDown(KeyCode.E)) { TryInteract(); } } void FixedUpdate() { rb.MovePosition(rb.position + moveInput.normalized * moveSpeed * Time.fixedDeltaTime); } void TryInteract() { float interactRadius = 1.2f; Collider2D[] colliders = Physics2D.OverlapCircleAll(transform.position, interactRadius); foreach (var col in colliders) { IInteractable interactable = col.GetComponent(); if (interactable != null) { interactable.Interact(this); ShowText(interactable.GetInteractionText()); break; } } } public void ShowText(string text) { if (uiTextDisplay != null) { StopAllCoroutines(); StartCoroutine(DisplayTextCoroutine(text)); } } private IEnumerator DisplayTextCoroutine(string text) { uiTextDisplay.text = text; yield return new WaitForSeconds(3.5f); uiTextDisplay.text = ""; } } public class GenericEnvironmentObject : MonoBehaviour, IInteractable { [TextArea] public string interactionMessage = "Es una ventana vieja..."; public void Interact(PlayerController player) { } public string GetInteractionText() => interactionMessage; } public class GrassComponent : MonoBehaviour { public GameObject breakingEffect; // Se rompe al posicionarse exactamente ENCIMA (Trigger) void OnTriggerEnter2D(Collider2D other) { if (other.CompareTag("Player")) { PlayerController player = other.GetComponent(); if (player != null) player.ShowText("*Crujido* Has pisado y roto la hierba."); if (breakingEffect != null) Instantiate(breakingEffect, transform.position, Quaternion.identity); Destroy(gameObject); } } } // ========================================== // 2. LA ROTONDA Y BLOQUEO DE CAMINOS // ========================================== public class CentralRoundabout : MonoBehaviour { public Transform ancientBlueTree; public bool[] pathsUnlocked = new bool[9]; // 9 caminos en total void Start() { // Solo 2 caminos desbloqueados al inicio de forma estricta for (int i = 0; i < 9; i++) pathsUnlocked[i] = false; pathsUnlocked[0] = true; // Camino a Ciudad Ígnea pathsUnlocked[1] = true; // Camino al Bosque Rowan } public bool TryEnterPath(int pathIndex, PlayerController player) { if (pathIndex < 0 || pathIndex >= 9) return false; if (pathsUnlocked[pathIndex]) { player.ShowText("Avanzas por el camino desbloqueado..."); return true; } else { player.ShowText("Este camino está bloqueado misteriosamente. No puedes avanzar."); return false; } } } // ========================================== // 3. CIUDAD ÍGNEA: ALDEANOS Y LLAVES // ========================================== public class VillagerNPC : MonoBehaviour, IInteractable { public bool isRelevant = false; [TextArea] public string loreMessage = ""; [TextArea] public string genericMessage = "Hola viajero, hace buen día."; public void Interact(PlayerController player) { } public string GetInteractionText() { if (isRelevant) return loreMessage; // Mensajes de suspense para los 4 elegidos return genericMessage; } } public class HiddenKeyObject : MonoBehaviour, IInteractable { public string keyID; public string objectName = "un viejo caldero"; public void Interact(PlayerController player) { if (!player.inventoryKeys.Contains(keyID)) { player.inventoryKeys.Add(keyID); } } public string GetInteractionText() => $"Rebuscas en {objectName}... ¡Has encontrado una llave oculta!"; } public class LockedHouse : MonoBehaviour, IInteractable { public string requiredKeyID; private bool isOpen = false; public void Interact(PlayerController player) { if (!isOpen && player.inventoryKeys.Contains(requiredKeyID)) { isOpen = true; player.inventoryKeys.Remove(requiredKeyID); } } public string GetInteractionText() { if (isOpen) return "La puerta está abierta. Entras a la casa."; return "La puerta está cerrada firmemente con llave. Debe estar oculta en algún lugar de la ciudad."; } } // ========================================== // 4. BOSQUE ROWAN: TALA Y CABAÑA // ========================================== public class GiantTree : MonoBehaviour, IInteractable { public bool isAncientBlueTree = false; public int woodAmount = 3; public void Interact(PlayerController player) { if (isAncientBlueTree) return; if (player.hasAxe && woodAmount > 0) { player.magicWood += woodAmount; woodAmount = 0; } } public string GetInteractionText() { if (isAncientBlueTree) return "Es el Árbol Azul Milenario. Una fuerza mística impide que le hagas el más mínimo rasguño."; if (!playerHasAxeReference()) return "Un árbol enorme de madera mágica. Necesitas un hacha para cortarlo."; if (woodAmount <= 0) return "El árbol ya ha sido talado."; return $"¡Usas el Hacha! Has obtenido Madera Mágica."; } private bool playerHasAxeReference() { var p = FindObjectOfType(); return p != null && p.hasAxe; } } public class RowanCabin : MonoBehaviour, IInteractable { private bool triggeredAnimation = false; public void Interact(PlayerController player) { if (!triggeredAnimation) { triggeredAnimation = true; TriggerCabinAnimation(); player.hasAxe = true; // Te da el hacha normal } } public string GetInteractionText() => "Llegas a la cabaña del bosque profunda. El leñador te entrega un Hacha Normal."; void TriggerCabinAnimation() { // Lógica para disparar la animación de llegada cinematográfica Debug.Log("Ejecutando animación cinemática de la Cabaña Rowan..."); } } // ========================================== // 5. ECONOMÍA Y TIENDAS // ========================================== public class IgneaShop : MonoBehaviour, IInteractable { public enum ShopType { BuyProperties, SellWood } public ShopType shopType; public void Interact(PlayerController player) { if (shopType == ShopType.SellWood && player.magicWood > 0) { int profit = player.magicWood * 15; player.gold += profit; player.magicWood = 0; } else if (shopType == ShopType.BuyProperties && player.gold >= 100) { player.gold -= 100; // Compra un Dojo de entrenamiento o acceso especial } } public string GetInteractionText() { if (shopType == ShopType.SellWood) return "Tienda: Has vendido toda tu Madera Mágica por monedas de oro."; return "Tienda Inmobiliaria: Compras accesos a dojos, casas vacías y lagos secretos por 100 de oro."; } } // ========================================== // 6. SISTEMA DE GUARDADO (5 RANURAS) // ========================================== [System.Serializable] public class SaveData { public int gold; public int magicWood; public bool hasAxe; public List keys; } public class SaveSystem : MonoBehaviour { public void SaveGame(int slot, PlayerController player) { if (slot < 1 || slot > 5) return; SaveData data = new SaveData { gold = player.gold, magicWood = player.magicWood, hasAxe = player.hasAxe, keys = new List(player.inventoryKeys) }; string json = JsonUtility.ToJson(data); PlayerPrefs.SetString($"JoyBitSave_Slot_{slot}", json); PlayerPrefs.Save(); Debug.Log($"Partida guardada y sobrescrita en la ranura: {slot}"); } public void LoadGame(int slot, PlayerController player) { string key = $"JoyBitSave_Slot_{slot}"; if (PlayerPrefs.HasKey(key)) { string json = PlayerPrefs.GetString(key); SaveData data = JsonUtility.FromJson(json); player.gold = data.gold; player.magicWood = data.magicWood; player.hasAxe = data.hasAxe; player.inventoryKeys = new List(data.keys); } } } }
